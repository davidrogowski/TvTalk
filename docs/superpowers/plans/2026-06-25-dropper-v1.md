# Dropper V1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship `tvtalk.fun/dropper` — paste a YouTube link, mark the beat drop, pick a quote from the TvTalk library, and the quote lands exactly as the beat drops (the "Two Friends" move), shareable by link, fully in-browser.

**Architecture:** A single static page (`dropper.html`) deployed with the rest of the site on Cloudflare — no backend. The YouTube IFrame Player API plays the song; we duck its volume during the spoken quote and snap it back to 100% at the drop. The quote is an existing `/audio/*.mp3` played via HTML5 `Audio`. All pure logic (URL parsing, timing math, share-link codec) lives in `dropper.timing.js` and is unit-tested with Node's built-in test runner.

**Tech Stack:** Plain HTML/CSS/JS (matches `index.html`, `clockthequote.html`); YouTube IFrame Player API; `node:test` + `node:assert` for unit tests; Playwright for headless layout screenshots; wrangler for deploy.

## Global Constraints

- **URL:** the app must work at `tvtalk.fun/dropper` (Cloudflare serves `dropper.html` at the extensionless path, like `clockthequote.html` → `/clockthequote`).
- **No backend / no new infra in V1.** No `yt-dlp`, no `ffmpeg`, no server-side audio. Pure static.
- **Reuse existing data:** quotes come from the global `shows` array in `shows.js` (each show: `{id, name, type, quotes:[{text, character, audioUrl}]}`). Audio is served from `/audio/...`; prefix `audioUrl` with `AUDIO_BASE = "/"`.
- **Timing defaults (exact):** `gap = 0` (quote ends ON the drop), `leadIn = 1.8`s, `duck = 20` (percent), `tail = 4`s. Nudge granularity: drop ±0.1s, gap ±0.5s.
- **Share params:** `?v=<videoId>&d=<dropTime>&q=<showId>:<clipIndex>&g=<gap>&dk=<duck>` (`g`/`dk` omitted when at default).
- **Committing in this repo:** plain `git add`/`git commit`/`git status` HANG (full-worktree scan over ~9,700 files on a slow vault FS). Use the `scripts/gcommit.sh` helper created in Task 1 — `bash scripts/gcommit.sh "message" file1 file2 …` — for every commit. Do NOT run bare `git commit`.
- **Files that ship vs. don't:** `dropper.html` and `dropper.timing.js` live at repo root (shipped). Tests and helpers go in `scripts/` (excluded by `.assetsignore`).

```js
// dropper.timing.js — pure logic for Dropper (URL parse, timing math, share codec).
// Dual-mode: browser global `DropperTiming` + CommonJS `module.exports`.
(function (root, factory) {
  const api = factory();
  if (typeof module !== 'undefined' && module.exports) module.exports = api;
  else root.DropperTiming = api;
})(typeof globalThis !== 'undefined' ? globalThis : this, function () {
  const ID = /^[A-Za-z0-9_-]{11}$/;

  function parseVideoId(input) {
    if (!input || typeof input !== 'string') return null;
    const s = input.trim();
    if (ID.test(s)) return s;
    let url;
    try { url = new URL(s); } catch (_) { return null; }
    const host = url.hostname.replace(/^www\./, '');
    if (host === 'youtu.be') {
      const id = url.pathname.slice(1, 12);
      return ID.test(id) ? id : null;
    }
    if (host === 'youtube.com' || host === 'm.youtube.com' || host === 'music.youtube.com') {
      if (url.pathname === '/watch') {
        const v = url.searchParams.get('v');
        return v && ID.test(v) ? v : null;
      }
      const m = url.pathname.match(/^\/(embed|shorts|v)\/([A-Za-z0-9_-]{11})/);
      if (m) return m[2];
    }
    return null;
  }

  return { parseVideoId };
});
```

```js
  function round3(n) { return String(Math.round(Number(n) * 1000) / 1000); }

  function encodeShare(state) {
    const p = new URLSearchParams();
    p.set('v', state.videoId);
    p.set('d', round3(state.dropTime));
    p.set('q', state.showId + ':' + state.clipIndex);
    if (state.gap !== undefined && state.gap !== DEFAULTS.gap) p.set('g', round3(state.gap));
    if (state.duck !== undefined && state.duck !== DEFAULTS.duck) p.set('dk', String(state.duck));
    return p.toString();
  }

  function decodeShare(search) {
    const p = new URLSearchParams(search || '');
    const v = p.get('v'), q = p.get('q');
    if (!v || !q) return null;
    const parts = q.split(':');
    if (parts.length !== 2) return null;
    const clipIndex = parseInt(parts[1], 10);
    if (Number.isNaN(clipIndex)) return null;
    const d = parseFloat(p.get('d'));
    return {
      videoId: v,
      showId: parts[0],
      clipIndex,
      dropTime: Number.isNaN(d) ? 0 : d,
      gap: p.has('g') ? parseFloat(p.get('g')) : DEFAULTS.gap,
      duck: p.has('dk') ? parseInt(p.get('dk'), 10) : DEFAULTS.duck,
    };
  }
```

```js
    return !!(state.videoId && state.dropMarked && state.quoteAudio && state.quoteDuration > 0);

    document.getElementById('momentCard').classList.remove('hidden');

  let momentTimer = null;
  function playMoment() {
    const warn = document.getElementById('momentWarn'); warn.textContent = '';
    const sch = DropperTiming.buildSchedule({
      dropTime: state.dropTime, quoteDuration: state.quoteDuration, gap: state.gap, duck: state.duck,
    });
    if (!sch.valid) { warn.textContent = sch.warning; return; }
    if (momentTimer) clearInterval(momentTimer);
    let quoteFired = false, dropFired = false;
    player.setVolume(100);
    player.seekTo(sch.playbackStart, true);
    player.playVideo();
    momentTimer = setInterval(() => {
      const t = player.getCurrentTime();
      if (!quoteFired && t >= sch.quoteStartAt) {
        quoteFired = true; player.setVolume(sch.duck);
        state.quoteAudio.currentTime = 0; state.quoteAudio.play();
      }
      if (!dropFired && t >= sch.dropAt) { dropFired = true; player.setVolume(100); }
      if (t >= sch.stopAt) {
        clearInterval(momentTimer); momentTimer = null;
        player.pauseVideo(); state.quoteAudio.pause();
      }
    }, 40);
  }
  document.getElementById('playBtn').addEventListener('click', playMoment);
```
